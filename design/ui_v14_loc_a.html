数据库 schema

CREATE TABLE building ( id INTEGER PRIMARY KEY, warehouse_id INTEGER REFERENCES warehouse(id), name TEXT NOT NULL -- 如 "1 号楼" ); CREATE TABLE floor ( id INTEGER PRIMARY KEY, building_id INTEGER REFERENCES building(id), name TEXT NOT NULL -- 如 "2 层" ); CREATE TABLE location ( id INTEGER PRIMARY KEY, floor_id INTEGER REFERENCES floor(id), code TEXT NOT NULL, -- 如 "A-01" zone_type TEXT ); -- 查询 "1 号楼总在仓"： SELECT SUM(inv.on_hand) FROM inventory inv JOIN location l ON l.id=inv.location_id JOIN floor f ON f.id=l.floor_id JOIN building b ON b.id=f.building_id WHERE b.name='1 号楼';
★ 优点
· 数据结构最严谨，3 层物理坐标完全可表达
· 按楼栋 / 楼层精细查询，SQL 直接 JOIN 一气呵成
· 适合 5+ 栋楼 / 10+ 层规模（多仓多层楼真实场景）
· 楼栋改名只改 1 处，不会出现"1栋 / 1号楼 / 1-栋"乱码
✗ 缺点
· 菜单 +2 项（楼栋号 / 楼层），20 人小团队认知负担重
· 建一个库位要点 3 次 picker，频繁操作慢
· 实装工作量大：5 张表（building/floor/location + admin/picker 各 2）
· 团队规模小时（每栋 1-3 层）属于过度设计
